function T_des = euler_XYpZ_to_T(x, y, z, alpha, beta, gamma)
    % XY''Z'' 欧拉角 → 齐次变换矩阵
    % R = Rz(gamma) * Ry(beta) * Rx(alpha)
    Rz_m = Rz(gamma);
    Ry_m = Ry(beta);
    Rx_m = Rx(alpha);
    T_des = eye(4);
    T_des(1:3,1:3) = Rz_m(1:3,1:3) * Ry_m(1:3,1:3) * Rx_m(1:3,1:3);
    T_des(1,4) = x;
    T_des(2,4) = y;
    T_des(3,4) = z;
end
