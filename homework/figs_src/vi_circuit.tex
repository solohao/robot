\documentclass[border=6pt]{standalone}
\usepackage{circuitikz}
\usepackage{amsmath}
\ctikzset{bipoles/thickness=1}
\begin{document}
\begin{circuitikz}[scale=1.05, transform shape, font=\small]

% ---------- Stage 1: non-inverting summing amplifier (U1) ----------
\node [op amp] (U1) at (3.0,4) {U1};
\coordinate (vp) at (1.5,3.6);     % V+ node (into U1 +)
% Vin -> Ra -> V+ node
\draw (-2.1,4.2) node[left]{$V_{in}\,(0\!\sim\!1\,\mathrm{V})$}
      to[short,-o] (-1.7,4.2);
\draw (-1.7,4.2) to[R,l=$R_a\,10\mathrm{k}$] (0.4,4.2) -- (0.4,3.6) -- (vp);
% Vref 0.25V -> Rb -> V+ node
\draw (-1.7,2.6) node[left]{$V_{ref}=0.25\,\mathrm{V}$}
      to[short,-o] (-1.7,2.6);
\draw (-1.7,2.6) to[R,l_=$R_b\,10\mathrm{k}$] (0.4,2.6) -- (0.4,3.6);
\node[circle,fill,inner sep=1pt] at (0.4,3.6) {};
\draw (vp) -- (U1.+);
\node at (0.75,3.9) {$V_+$};
% inverting input: Rg to ground, Rf feedback
\coordinate (vm) at (1.5,4.4);
\draw (vm) -- (U1.-);
\draw (vm) -- (0.4,4.4) to[R,l_=$R_g\,10\mathrm{k}$] (0.4,5.6) node[ground,rotate=180]{};
\node[circle,fill,inner sep=1pt] at (vm) {};
\draw (vm) -- (1.5,6.0) to[R,l=$R_f\,10\mathrm{k}$] (4.6,6.0) -- (4.6,4) -- (U1.out);
% output = control voltage Vc
\draw (U1.out) -- (5.6,4);
\node at (5.95,4.35) {$V_c$};

% ---------- Stage 2: V/I converter (U2 + Q1 + Rs) ----------
\node [op amp] (U2) at (7.8,3.2) {U2};
\draw (5.6,4) -- (5.6,3.5) -- (U2.+);      % Vc -> U2 +
% MOSFET Q1 (N-channel) driven by U2 output
\node [nigfete] (Q1) at (9.9,3.2) {Q1};
\draw (U2.out) -- (Q1.G);
% drain to load and +24V
\draw (Q1.D) -- (9.9,5.0) to[R,l=$R_L$] (9.9,6.2) -- (11.5,6.2)
      node[right]{$+24\,\mathrm{V}$} to[short,-o] (11.8,6.2);
% source -> sense resistor Rs -> ground; feedback to U2 -
\draw (Q1.S) -- (9.9,1.6) coordinate(sns);
\draw (sns) to[R,l_=$R_s\,62.5\,\Omega$] (9.9,0.2) node[ground]{};
\draw (U2.-) -- (6.9,2.6) -- (6.9,1.6) -- (sns);
\node[circle,fill,inner sep=1pt] at (sns) {};
\node at (10.3,1.0) {\scriptsize $V_{R_s}$};

% current arrow
\draw[-{Latex}] (10.25,5.4) -- (10.25,4.6) node[midway,right]{$I_{out}$};

\end{circuitikz}
\end{document}
