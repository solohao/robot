% shared ladder-drawing macros (TikZ)
\usetikzlibrary{arrows.meta,calc}
% NO contact centered at (#1,#2), label #3 above.  Total width 1.0
\newcommand{\noC}[3]{%
  \draw (#1-0.5,#2)--(#1-0.16,#2);
  \draw (#1+0.16,#2)--(#1+0.5,#2);
  \draw (#1-0.16,#2-0.26)--(#1-0.16,#2+0.26);
  \draw (#1+0.16,#2-0.26)--(#1+0.16,#2+0.26);
  \node[above] at (#1,#2+0.28){\scriptsize #3};
}
% NC contact (with slash)
\newcommand{\ncC}[3]{%
  \draw (#1-0.5,#2)--(#1-0.16,#2);
  \draw (#1+0.16,#2)--(#1+0.5,#2);
  \draw (#1-0.16,#2-0.26)--(#1-0.16,#2+0.26);
  \draw (#1+0.16,#2-0.26)--(#1+0.16,#2+0.26);
  \draw (#1-0.16,#2-0.26)--(#1+0.16,#2+0.26);
  \node[above] at (#1,#2+0.28){\scriptsize #3};
}
% output coil ( ), label above, optional letter inside via #4
\newcommand{\coilX}[4]{%
  \draw (#1-0.5,#2)--(#1-0.26,#2);
  \draw (#1+0.26,#2)--(#1+0.5,#2);
  \draw (#1-0.26,#2) arc (150:210:0.52);
  \draw (#1+0.26,#2) arc (30:-30:0.52);
  \node at (#1,#2){\scriptsize #4};
  \node[above] at (#1,#2+0.30){\scriptsize #3};
}
\newcommand{\coil}[3]{\coilX{#1}{#2}{#3}{}}
% TON on-delay timer box centered at (#1,#2); #3 name, #4 preset
\newcommand{\tonbox}[4]{%
  \draw (#1-0.85,#2-0.55) rectangle (#1+0.85,#2+0.55);
  \node at (#1,#2+0.28){\scriptsize #3};
  \node at (#1,#2-0.02){\scriptsize \textbf{TON}};
  \node at (#1,#2-0.32){\scriptsize #4};
  \draw (#1-0.85,#2)--(#1-1.2,#2);   % enable input stub
}
